/**
 * Firestore Security Rules for Team Chat
 * Copy these rules into your Firebase Console under Firestore Database > Rules
 *
 * IMPORTANT: Merge these with your existing rules, don't replace everything!
 */

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper function: Check if user is a team member
    function isTeamMember(teamId) {
      let team = get(/databases/$(database)/documents/teams/$(teamId));
      return request.auth != null &&
             request.auth.token.email in team.data.memberEmails;
    }

    // Helper function: Check if team has active subscription (can access chat)
    function hasActiveSubscription(teamId) {
      let team = get(/databases/$(database)/documents/teams/$(teamId));
      return team.data.billing.subscription.status == 'active';
    }

    // ========================================================================
    // CHANNELS
    // ========================================================================

    match /teams/{teamId}/channels/{channelId} {
      // Allow read if user is team member with active subscription
      allow read: if isTeamMember(teamId) && hasActiveSubscription(teamId);

      // Allow create if user is team member with active subscription
      allow create: if isTeamMember(teamId) &&
                       hasActiveSubscription(teamId) &&
                       request.resource.data.createdBy == request.auth.token.email;

      // Allow update if user is team member with active subscription
      allow update: if isTeamMember(teamId) && hasActiveSubscription(teamId);

      // Allow delete if user is team owner or admin
      allow delete: if isTeamMember(teamId) &&
                       hasActiveSubscription(teamId) &&
                       (request.auth.token.email == get(/databases/$(database)/documents/teams/$(teamId)).data.owner ||
                        get(/databases/$(database)/documents/teams/$(teamId)).data.members[request.auth.token.email].role in ['admin', 'owner']);
    }

    // ========================================================================
    // MESSAGES
    // ========================================================================

    match /teams/{teamId}/messages/{channelId}/items/{messageId} {
      // Allow read if user is team member with active subscription
      allow read: if isTeamMember(teamId) && hasActiveSubscription(teamId);

      // Allow create if user is team member with active subscription
      // and sender matches authenticated user
      allow create: if isTeamMember(teamId) &&
                       hasActiveSubscription(teamId) &&
                       request.resource.data.senderEmail == request.auth.token.email;

      // Allow update if user is the message sender (for editing own messages)
      allow update: if isTeamMember(teamId) &&
                       hasActiveSubscription(teamId) &&
                       resource.data.senderEmail == request.auth.token.email;

      // Allow delete if user is the message sender or team admin
      allow delete: if isTeamMember(teamId) &&
                       hasActiveSubscription(teamId) &&
                       (resource.data.senderEmail == request.auth.token.email ||
                        get(/databases/$(database)/documents/teams/$(teamId)).data.members[request.auth.token.email].role in ['admin', 'owner']);
    }

    // ========================================================================
    // UNREAD TRACKING (Optional - for future implementation)
    // ========================================================================

    match /teams/{teamId}/unread/{userId} {
      // Users can only read/write their own unread tracking
      allow read, write: if isTeamMember(teamId) &&
                            hasActiveSubscription(teamId) &&
                            userId == request.auth.uid;
    }

    // ========================================================================
    // TYPING INDICATORS (Optional - for future implementation)
    // ========================================================================

    match /teams/{teamId}/typing/{typingDoc} {
      // Team members can read/write typing indicators
      allow read, write: if isTeamMember(teamId) &&
                            hasActiveSubscription(teamId);
    }
  }
}

 * DEPLOYMENT INSTRUCTIONS:
 *
 * 1. Go to Firebase Console > Firestore Database > Rules
 * 2. Find your existing rules for /teams/{teamId}
 * 3. Add the channels and messages rules above INSIDE the teams/{teamId} match block
 * 4. Click "Publish" to deploy the rules

 *
